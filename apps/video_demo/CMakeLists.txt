set(BGMEDIA_TOK_FIXTURE_ROOT "" CACHE PATH "BG Tok checkout for the optional Linux fixture demo")
set(BGMEDIA_TOK_FIXTURE_LIB "" CACHE FILEPATH "Prebuilt bg-tok-fixture-ffi static library")
if(BGMEDIA_TOK_FIXTURE_ROOT OR BGMEDIA_TOK_FIXTURE_LIB)
    if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT TARGET bgmedia_video_cache OR
       NOT EXISTS "${BGMEDIA_TOK_FIXTURE_ROOT}/crates/bg-tok-fixture-ffi/include/bg_tok_fixture.h" OR
       NOT EXISTS "${BGMEDIA_TOK_FIXTURE_LIB}")
        message(FATAL_ERROR "BG Tok demo requires Linux, VIDEO_CACHE, fixture headers and a prebuilt fixture static library")
    endif()
    target_include_directories(bgmedia_video_demo PRIVATE "${BGMEDIA_TOK_FIXTURE_ROOT}/crates/bg-tok-fixture-ffi/include")
    target_compile_definitions(bgmedia_video_demo PRIVATE BGMEDIA_TOK_FIXTURE=1)
    target_link_libraries(bgmedia_video_demo PRIVATE BGMedia::VideoCache "${BGMEDIA_TOK_FIXTURE_LIB}" ${CMAKE_DL_LIBS} m)
endif()

set(BGMEDIA_TOK_EXPLORE_ROOT "" CACHE PATH "BG Tok integration checkout for the live probe")
set(BGMEDIA_TOK_EXPLORE_LIB "" CACHE FILEPATH "Prebuilt bg-tok-explore-ffi static library")
option(BGMEDIA_ENABLE_LIVE_TIKTOK_TESTS "Enable opt-in network-dependent TikTok tests" OFF)
if(BGMEDIA_TOK_EXPLORE_ROOT OR BGMEDIA_TOK_EXPLORE_LIB)
    if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR
       NOT EXISTS "${BGMEDIA_TOK_EXPLORE_ROOT}/crates/bg-tok-explore-ffi/include/bg_tok_explore.h" OR
       NOT EXISTS "${BGMEDIA_TOK_EXPLORE_LIB}")
        message(FATAL_ERROR "TikTok live probe requires Linux, Explore headers and a prebuilt static library")
    endif()
    add_executable(bgmedia_tiktok_probe TikTokProbe.cpp)
    target_include_directories(bgmedia_tiktok_probe PRIVATE "${BGMEDIA_TOK_EXPLORE_ROOT}/crates/bg-tok-explore-ffi/include")
    target_link_libraries(bgmedia_tiktok_probe PRIVATE BGMedia::VideoFFmpeg "${BGMEDIA_TOK_EXPLORE_LIB}" ${CMAKE_DL_LIBS} m)
    target_compile_options(bgmedia_tiktok_probe PRIVATE ${BGMEDIA_WARNING_FLAGS})
    if(BGMEDIA_ENABLE_LIVE_TIKTOK_TESTS AND BGMEDIA_BUILD_TESTS)
        add_test(NAME bgmedia.tiktok_live_probe COMMAND bgmedia_tiktok_probe)
        set_tests_properties(bgmedia.tiktok_live_probe PROPERTIES TIMEOUT 120 LABELS "live;integration")
    endif()
endif()
